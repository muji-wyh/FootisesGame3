# 214+LK Upper-KO Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn Blaze/Ultron `214+LK` into one heavy-impact kick that launches the victim with `KB_UpperKO_Flip`.

**Architecture:** Keep the change inside Blaze's existing `MoveData`; Ultron inherits it through `Blaze.build()`. Reuse the existing authored hit-reaction override and deterministic combat harness, with no new engine fields or animation special cases.

**Tech Stack:** Godot 4.7, typed GDScript, fixed 60 Hz simulation, Kubold animation library.

## Global Constraints

- `214+LK` connects exactly once.
- The airborne victim reaction is `KB_UpperKO_Flip`.
- Preserve the old move's total damage (`48`) and meter gain (`16`).
- Keep startup, total 34-frame commitment, and `super_inferno` cancel unchanged.
- End the hitbox on the upward-kick impact frame; do not leave it active during retraction.
- Keep the corrected vertical kick hitbox aligned with the measured impact pose.
- Do not alter global upper-knockdown animation selection.
- Ultron must inherit the same behavior without Ultron-specific code.

	_check("Ember Lift is a grounded single-hit launcher",

After the existing input-recognition context in `_test_ember_lift()`, add a whiff-lifetime
check followed by the direct-hit check:

```gdscript
	var whiff := _build()
	var whiff_fighter: Fighter = whiff["f1"]
	_p1_qcb(whiff, GameConst.Btn.LK)
	var live_hitbox_ticks := 0
	for i in range(move.total_frames() + 2):
		if whiff_fighter.has_active_hitbox():
			live_hitbox_ticks += 1
		_step(whiff, _neutral(), _neutral(), 1)
	_check("whiffed 214 + LK hitbox ends with the upward kick", live_hitbox_ticks == 1)
	whiff["arena"].queue_free()

	var launch := _build()
	var attacker: Fighter = launch["f1"]
	var victim: Fighter = launch["f2"]
	attacker.position.x = -0.34
	victim.position.x = 0.34
	_p1_qcb(launch, GameConst.Btn.LK)
	var max_hits := 0
	var saw_flip := false
	var saw_violent_launch := false
	for i in range(90):
		_step(launch, _neutral(), _neutral(), 1)
		max_hits = maxi(max_hits, victim.combo_count)
		if victim.combo_count == 1:
			saw_flip = saw_flip or victim.hit_reaction_clip == "KB_UpperKO_Flip"
			saw_violent_launch = saw_violent_launch or (
				victim.velocity.y >= 7.5 and absf(victim.velocity.x) >= 11.0)
	_check("direct 214 + LK connects exactly once", max_hits == 1)
	_check("direct 214 + LK records the upper-KO flip", saw_flip)
	_check("direct 214 + LK violently launches the victim", saw_violent_launch)
	launch["arena"].queue_free()

	var guarded := _build()
	var guard_attacker: Fighter = guarded["f1"]
	var guard_victim: Fighter = guarded["f2"]
	guard_attacker.position.x = -0.34
	guard_victim.position.x = 0.34
	var guard_health := guard_victim.health
	_p1_qcb(guarded, GameConst.Btn.LK)
	var blocked := false
	var block_velocity := 0.0
	var attacker_recoil := 0.0
	for i in range(30):
		_step(guarded, _neutral(), _mk(1, 0), 1)
		if guard_victim.state == Fighter.State.BLOCKSTUN:
			blocked = true
			block_velocity = absf(guard_victim.velocity.x)
			attacker_recoil = guard_attacker._recoil_vel
			break
	_check("guarded 214 + LK creates an intentional spacing reset",
		blocked
		and guard_victim.health == guard_health
		and not guard_victim.launched
		and is_equal_approx(block_velocity, 5.5)
		and attacker_recoil < 0.0)
	guarded["arena"].queue_free()
```
